```sql
[Id]           INT IDENTITY PRIMARY KEY
[RecordedAt]   DATETIME2  NOT NULL  DEFAULT GETUTCDATE()

[TotalValue]   DECIMAL(18,4) NOT NULL
[StocksValue]  DECIMAL(18,4) NOT NULL
[CashValue]    DECIMAL(18,4) NOT NULL
```
